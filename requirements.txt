# Ядро НЕ использует трансформеры для рассуждений, но SDR кодируется
# ЭМБЕДДИНГАМИ (core/encoder.py). Значит:
#
#   би́ты SDR -> пороги узнавания (13/17, шумовой пол, unknown_threshold)
#
# Пороги — это ЗАМЕР на конкретных битах. У другой версии sentence-transformers
# или transformers эмбеддинги ДРУГИЕ, а код об этом не догадается: SDR останется
# «семантическим», overlap останется «в норме», и вскроется это только на живом
# ответе. Поэтому версии здесь зафиксированы (==), а не «примерно так», и поэтому
# же существует tests/golden_bits.json — он ловит расхождение побитово.
#
# ПРАВИЛО: любое изменение ЭТОГО файла = новый прогон замеров
# (experiments/RESULTS.md) и осознанное обновление tests/golden_bits.json.
# Тест tests/test_golden_bits.py сверяет 17 слов побитово и краснеет ПЕРВЫМ,
# ещё до того, как расхождение уйдёт в живые ответы.
#
# Установка на чистой машине:
#     python3.12 -m venv .venv
#     .venv/bin/pip install -r requirements.txt
# Проверка:
#     .venv/bin/python -m unittest discover -s tests
#     .venv/bin/python tests/test_golden_bits.py      # би́ты обязаны совпасть
#
# Модель весов НЕ в pip: paraphrase-multilingual-MiniLM-L12-v2 (~449 МБ)
# скачивается sentence-transformers при первом запуске в HF-кэш.

# ── ЯДРО КОДИРОВАНИЯ SDR ───────────────────────────────────────────────

# чтобы pip не притащил версию, несовместимую с зафиксированным numpy 2.x.
scikit-learn==1.9.1
